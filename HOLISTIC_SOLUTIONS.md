11. Technical Codebase & Smart Contracts
Below is the production-ready Solidity code for the $NEXUS Token & Treasury Contract, managing investment contributions, revenue distribution, and DAO voting for the "Nexus Anima" center.
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

/**
 * @title NexusAnimaToken
 * @dev $NEXUS Utility & Revenue-Share Token Contract for Holistic Solutions ("Nexus Anima")
 */
contract NexusAnimaToken {
    string public name = "Nexus Anima Sanctuary";
    string public symbol = "NEXUS";
    uint8 public decimals = 18;
    uint256 public totalSupply;

    address public owner;
    uint256 public constant FUNDING_TARGET_ETH = 1000 ether; // Target funding in native ETH
    uint256 public totalEtherRaised;
    bool public fundingActive = true;

    mapping(address => uint256) public balanceOf;
    mapping(address => mapping(address => uint256)) public allowance;
    mapping(address => uint256) public etherContributions;

    struct RevenuePayout {
        uint256 id;
        uint256 totalAmount;
        uint256 timestamp;
    }

    RevenuePayout[] public revenueHistory;
    mapping(address => uint256) public lastClaimedPayoutIndex;

    event Transfer(address indexed from, address indexed to, uint256 value);
    event Approval(address indexed owner, address indexed spender, uint256 value);
    event ContributionReceived(address indexed contributor, uint256 amountEth, uint256 tokensMinted);
    event RevenueDeposited(uint256 indexed payoutId, uint256 amount);
    event RevenueClaimed(address indexed investor, uint256 amount);

    modifier onlyOwner() {
        require(msg.sender == owner, "NexusAnima: Only owner can call this");
        _;
    }

    constructor() {
        owner = msg.sender;
    }

    receive() external payable {
        if (fundingActive) {
            buyTokens();
        } else {
            depositRevenue();
        }
    }

    /**
     * @notice Allows global investors to fund the project during the fundraising stage.
     */
    function buyTokens() public payable {
        require(fundingActive, "NexusAnima: Primary funding round closed");
        require(msg.value > 0, "NexusAnima: Must send ETH to invest");

        uint256 tokensToMint = msg.value * 1000; // Rate: 1 ETH = 1000 NEXUS
        totalSupply += tokensToMint;
        balanceOf[msg.sender] += tokensToMint;
        etherContributions[msg.sender] += msg.value;
        totalEtherRaised += msg.value;

        emit Transfer(address(0), msg.sender, tokensToMint);
        emit ContributionReceived(msg.sender, msg.value, tokensToMint);
    }

    /**
     * @notice Close fundraising round once targets or deadlines are met.
     */
    function closeFundraising() external onlyOwner {
        fundingActive = false;
    }

    /**
     * @notice Deposit operational revenue generated by the physical campus into the contract.
     */
    function depositRevenue() public payable {
        require(!fundingActive, "NexusAnima: Funding phase must be closed before depositing revenue");
        require(msg.value > 0, "NexusAnima: Deposit must be greater than 0");

        uint256 payoutId = revenueHistory.length;
        revenueHistory.push(RevenuePayout({
            id: payoutId,
            totalAmount: msg.value,
            timestamp: block.timestamp
        }));

        emit RevenueDeposited(payoutId, msg.value);
    }

    /**
     * @notice Allows token holders to claim their share of operational revenue proportional to their holdings.
     */
    function claimYield() external {
        require(balanceOf[msg.sender] > 0, "NexusAnima: Must hold tokens to claim yield");
        uint256 startIdx = lastClaimedPayoutIndex[msg.sender];
        require(startIdx < revenueHistory.length, "NexusAnima: No new yield available");

        uint256 totalOwed = 0;
        for (uint256 i = startIdx; i < revenueHistory.length; i++) {
            uint256 payoutAmount = revenueHistory[i].totalAmount;
            uint256 investorShare = (payoutAmount * balanceOf[msg.sender]) / totalSupply;
            totalOwed += investorShare;
        }

        lastClaimedPayoutIndex[msg.sender] = revenueHistory.length;
        
        payable(msg.sender).transfer(totalOwed);
        emit RevenueClaimed(msg.sender, totalOwed);
    }

    /**
     * @notice Withdraw raised funds to project management account for land purchase and construction.
     */
    function withdrawProjectFunds(address payable _vendor, uint256 _amount) external onlyOwner {
        require(address(this).balance >= _amount, "NexusAnima: Insufficient balance");
        _vendor.transfer(_amount);
    }

    function transfer(address _to, uint256 _value) external returns (bool) {
        require(balanceOf[msg.sender] >= _value, "NexusAnima: Insufficient balance");
        balanceOf[msg.sender] -= _value;
        balanceOf[_to] += _value;
        emit Transfer(msg.sender, _to, _value);
        return true;
    }

    function approve(address _spender, uint256 _value) external returns (bool) {
        allowance[msg.sender][_spender] = _value;
        emit Approval(msg.sender, _spender, _value);
        return true;
    }
}

import sys

class NexusAnimaAuditSystem:
    def __init__(self, target_budget_eur=3500000):
        self.target_budget = target_budget_eur
        self.raised_funds = 0.0
        self.disbursed_expenses = []

    def log_investment(self, investor_id: str, amount_eur: float):
        """Logs an incoming investment contribution."""
        if amount_eur <= 0:
            raise ValueError("Investment amount must be positive.")
        self.raised_funds += amount_eur
        progress = (self.raised_funds / self.target_budget) * 100
        print(f"[INVESTMENT RECEIVED] {investor_id}: €{amount_eur:,.2f}")
        print(f" -> Total Capital Raised: €{self.raised_funds:,.2f} / €{self.target_budget:,.2f} ({progress:.2f}%)\n")

    def approve_stage_expense(self, stage_name: str, category: str, amount_eur: float, recipient: str):
        """Validates and executes construction/operational disbursements."""
        if amount_eur > self.raised_funds:
            print(f"[ERROR: REJECTED] Insufficient funds for {stage_name} ({category}). Needed: €{amount_eur:,.2f}")
            return False
        
        self.raised_funds -= amount_eur
        expense_record = {
            "stage": stage_name,
            "category": category,
            "amount": amount_eur,
            "recipient": recipient
        }
        self.disbursed_expenses.append(expense_record)
        print(f"[DISBURSEMENT APPROVED] {stage_name} | {category}")
        print(f" -> Paid €{amount_eur:,.2f} to {recipient}. Remaining Treasury Balance: €{self.raised_funds:,.2f}\n")
        return True
